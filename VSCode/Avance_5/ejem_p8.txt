SELECT C.M#, MNOMBRE
FROM MATERIAS, (SELECT M#
                FROM INSCRITOS
                WHERE A# IN (SELECT A#
                             FROM ALUMNOS
                             WHERE ANOMBRE='MESSI LIONEL')) E
WHERE MATERIAS.M#=C.M#
